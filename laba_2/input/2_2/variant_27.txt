# Задание 2.2, вариант 27 (a = 3):
#   a*x1^2 - x2 + x2^2 - a = 0
#   x1 - sqrt(x2 + a) + 1 = 0
eps = 0.0001
a = 3
f1 = a*x1^2 - x2 + x2^2 - a
f2 = x1 - sqrt(x2 + a) + 1
# Матрица Якоби для метода Ньютона: J_ij = df_i/dx_j
J11 = 2*a*x1
J12 = 2*x2 - 1
J21 = 1
J22 = -1/(2*sqrt(x2 + a))
# Метод простой итерации: если выражать x1, x2 из уравнений напрямую, ||phi'|| > 1.
# Поэтому phi(x) = x - L*f(x), где L = J(x(0))^-1 (округлено) — обратная матрица
# Якоби в начальном приближении; тогда phi'(x) = E - L*J(x) близка к нулю около x(0).

phi2 = x2 - (0.353*f1 - 2.332*f2)
